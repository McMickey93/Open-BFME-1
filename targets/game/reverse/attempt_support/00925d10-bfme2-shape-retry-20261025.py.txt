from pathlib import Path
import sys, json, re, contextlib, time, itertools
sys.path.insert(0, str(Path('tools').resolve()))
import build
from experiment_store import compile_cached, record_result

directory = Path(__file__).resolve().parent
base = (directory / 'base.cpp').read_text()
symbol = '?bfmeSet1016@BfmeDst1016@@QAEXHHH@Z'
image = Path(build.EXE).read_bytes()
offset = build.rva_to_file_offset(build.pe_sections(image), 0x00925D10)
retail = image[offset:offset+1634]
best = 20
results = directory / 'results.jsonl'

def measure(name, source):
    global best
    path = directory / (name + '.cpp')
    path.write_text(source)
    with (directory / (name + '.log')).open('w') as log:
        try:
            with contextlib.redirect_stdout(log), contextlib.redirect_stderr(log):
                obj, cached = compile_cached(path)
            compiled, relocs = build.read_object_symbol_bytes(obj, symbol)
            a, b = bytearray(compiled), bytearray(retail)
            for at, kind, _ in relocs:
                width = build.RELOC_WIDTH[kind]
                a[at:at+width] = bytes(width)
                b[at:at+width] = bytes(width)
            diffs = [i for i in range(min(len(a),len(b))) if a[i] != b[i]]
            count = len(diffs) + abs(len(a)-len(b))
            record_result(path, symbol, 0x00925D10, retail, compiled, relocs)
            result = dict(name=name, size=len(a), differences=count, offsets=diffs,
                          relocations=relocs, object=str(obj))
            if count < best:
                best = count
                print('BEST',name,'size',len(a),'differences',count,flush=True)
        except (Exception, SystemExit) as error:
            result = dict(name=name,error=str(error))
    with results.open('a') as output:
        output.write(json.dumps(result)+'\n')
    return result

mode = sys.argv[1]
if mode == 'parens':
    lines = base.splitlines()
    indexes = [i for i,line in enumerate(lines) if re.match(r'\s*po->[XYZ] = ',line)]
    assert len(indexes) == 6
    for mask in range(1,64):
        changed = lines[:]
        for bit, i in enumerate(indexes):
            if mask & (1 << bit):
                left,right = changed[i].split(' = ',1)
                changed[i] = left+' = ('+right[:-1]+');'
        measure('parens-%02d' % mask, '\n'.join(changed)+'\n')
        if mask % 8 == 0:
            print('finished',mask,'paren masks; best',best,flush=True)
elif mode == 'refs':
    start = base.index('\t\t\twhile (c--) {',base.index('\t\t} else {'))
    end = base.index('\t\t\t\t++in;', start)
    normal = base[start:end]
    for row in range(3):
        for col, field in enumerate('XYZ'):
            for slot in range(row+1):
                lines = normal.splitlines()
                needle = 'm[%d].%s' % (row,field)
                lines = [line.replace(needle,'coefficient') for line in lines]
                lines.insert(1+slot,'\t\t\t\tconst float &coefficient = '+needle+';')
                measure('ref-m%d%d-slot%d' % (row,col,slot),
                        base[:start]+'\n'.join(lines)+'\n'+base[end:])
    for col, field in enumerate('XYZ'):
        for slot in range(3):
            lines = normal.splitlines()
            for i in range(1+slot,len(lines)):
                lines[i] = lines[i].replace('in->'+field,'sample')
            lines.insert(1+slot,'\t\t\t\tconst float &sample = in->'+field+';')
            measure('ref-v%d-slot%d' % (col,slot),
                    base[:start]+'\n'.join(lines)+'\n'+base[end:])
    print('finished coefficient and sample reference trials; best',best,flush=True)
elif mode == 'volatile':
    start = base.index('\t\t\twhile (c--) {',base.index('\t\t} else {'))
    end = base.index('\t\t\t\t++in;', start)
    normal = base[start:end]
    for row in range(3):
        for col, field in enumerate('XYZ'):
            for side in ['matrix','sample']:
                lines = normal.splitlines()
                needle = 'm[%d].%s' % (row,field) if side == 'matrix' else 'in->'+field
                lines[1+row] = lines[1+row].replace(needle,'*(const volatile float *)&'+needle)
                measure('volatile-%s%d%d' % (side,row,col),
                        base[:start]+'\n'.join(lines)+'\n'+base[end:])
    print('finished volatile read trials; best',best,flush=True)
elif mode == 'inline':
    start = base.index('\t\t\tin = srcN[0];',base.index('\t\t} else {'))
    end = base.index('\t\t}\n\t\tlinks += n;',start)
    for pointer in [False,True]:
        declarations = ['const Matrix3D '+('*pm' if pointer else '&m'),
                        'const Vector3 *in','Vector3 *out','int c']
        arguments = ['pm' if pointer else 'm','srcN[0]','dstN','n']
        for order in itertools.permutations(range(4)):
            helper = 'static __forceinline void rva00925D10Normals('+','.join(declarations[i] for i in order)+')\n{\n'
            if pointer:
                helper += 'const Matrix3D &m=*pm;\n'
            helper += '''while(c--) {
out->X=m[0].X*in->X+m[0].Y*in->Y+m[0].Z*in->Z;
out->Y=m[1].X*in->X+m[1].Y*in->Y+m[1].Z*in->Z;
out->Z=m[2].X*in->X+m[2].Y*in->Y+m[2].Z*in->Z;
++in;
++out;
}
}
'''
            source = base[:start]+'\t\t\trva00925D10Normals('+','.join(arguments[i] for i in order)+');\n'+base[end:]
            at=source.index('void BfmeDst1016::bfmeSet1016')
            source=source[:at]+helper+'\n'+source[at:]
            measure('inline-'+('p' if pointer else 'r')+''.join(map(str,order)),source)
    print('finished inline parameter-context trials; best',best,flush=True)
