def trial(name,src):

  with (root/(name+'.compile.log')).open('w') as f,contextlib.redirect_stdout(f),contextlib.redirect_stderr(f):

  diffs=[i for i in range(min(len(a),len(b))) if a[i]!=b[i]]
  row=dict(name=name,size=len(data),diffs=len(diffs)+abs(len(data)-1036),first=hex(diffs[0]) if diffs else '',offsets=[hex(i) for i in diffs],object=str(obj),relocs=len(relocs))
  experiment_store.record_result(p,symbol,0x4e8320,target,data,relocs)
